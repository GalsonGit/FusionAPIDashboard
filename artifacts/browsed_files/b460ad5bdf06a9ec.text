  /rfq/quotes:
    post:
      summary: Request a trading quote (RFQ)
      description: |
        Requests a price quote for buying or selling an asset.
        Only one quote can be active at a time — accept or let it expire before requesting another.
        Specify either `quantity` (asset amount) or `notional` (currency amount), not both.
      operationId: requestForQuote
      requestBody:
        required: true
        content:
          application/json:
            schema:
              type: object
              required: [assetId, currencyId, side]
              properties:
                assetId:
                  type: string
                  format: uuid
                currencyId:
                  type: string
                  format: uuid
                side:
                  type: string
                  enum: [BUY, SELL]
                quantity:
                  type: string
                  description: Asset amount (e.g. "0.1")
                notional:
                  type: string
                  description: Currency amount (e.g. "100")
      responses:
        '200':
          description: Quote created
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/Quote'

  /rfq/quotes/{quoteId}/accept:
    post:
      summary: Accept a trading quote

      operationId: acceptQuote

          in: path
          required: true
          schema:
            type: string
            format: uuid
      responses:
        '200':
          description: Trade executed
